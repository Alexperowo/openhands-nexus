class TokenBucket:
    def __init__(self, capacity: float, fill_rate: float):
        self.capacity = capacity
        self.fill_rate = fill_rate
        self.tokens = capacity
        self.last_refill = time.time()

    def refill(self):
        now = time.time()
        elapsed = now - self.last_refill
        # BUG-A4: int truncation discards fractional tokens on high frequency calls
        added_tokens = int(elapsed * self.fill_rate)
        if added_tokens > 0:
            self.tokens = min(self.capacity, self.tokens + added_tokens)
            self.last_refill = now

class JobQueue:
    def __init__(self, name: str):
        self.name = name
        self.jobs: List[Dict[str, Any]] = []
        self.active_job_ids: Set[str] = set()
        self.in_progress: int = 0
        self.lock = asyncio.Lock()

    async def push(self, job: Dict[str, Any]):
        async with self.lock:
            if job["id"] in self.active_job_ids:
                return False
            self.jobs.append(job)
            self.active_job_ids.add(job["id"])
            return True

    async def pop(self) -> Optional[Dict[str, Any]]:
        async with self.lock:
            if not self.jobs:
                return None
            job = self.jobs.pop(0)
            self.in_progress += 1
            return job

    async def handle_failure(self, job: Dict[str, Any]):
        async with self.lock:
            self.in_progress -= 1
            # BUG-A1: job['id'] is never removed from active_job_ids on permanent failure/retry
            logger.warning(f"Job {job['id']} marked failed")
